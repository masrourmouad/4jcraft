163HRESULT CScene_HowToPlay::OnKeyDown(XUIMessageInput* pInputData,
164 BOOL& rfHandled) {
165 ui.AnimateKeyPress(pInputData->UserIndex, pInputData->dwKeyCode);
166
167 // Explicitly handle B button presses
168 switch (pInputData->dwKeyCode) {
169 case VK_PAD_B:
170 case VK_ESCAPE: {
171 app.NavigateBack(pInputData->UserIndex);
172 rfHandled = TRUE;
173 } break;
174 case VK_PAD_A: {
175 // Next page
176 int iNextPage = (int)(m_eCurrPage) + 1;
177 if (iNextPage != eHowToPlay_NumPages) {
178 StartPage((EHowToPlayPage)(iNextPage));
179 CXuiSceneBase::PlayUISFX(eSFX_Press);
180 }
181 rfHandled = TRUE;
182 } break;
183 case VK_PAD_X: {
184 // Next page
185 int iPrevPage = (int)(m_eCurrPage)-1;
186 if (iPrevPage >= 0) {
187 StartPage((EHowToPlayPage)(iPrevPage));
188 CXuiSceneBase::PlayUISFX(eSFX_Press);
189 }
190 rfHandled = TRUE;
191 } break;
192 }
193 return S_OK;
194}

196void CScene_HowToPlay::StartPage(EHowToPlayPage ePage) {
197 int iBaseSceneUser;
198 // if we're not in the game, we need to use basescene 0
199 if (Minecraft::GetInstance()->level == NULL) {
200 iBaseSceneUser = DEFAULT_XUI_MENU_USER;
201 } else {
202 iBaseSceneUser = m_iPad;
203 }
204 m_eCurrPage = ePage;
205
206 // Turn off everything.
207 for (int i = 0; i < eHowToPlay_NumTexts; ++i) {
208 m_aTextControls[i].SetShow(FALSE);
209 }
210 for (int i = 0; i < eHowToPlay_NumImages; ++i) {
211 m_aImageControls[i].SetShow(FALSE);
212 }
213 for (int i = 0; i < eHowToPlay_NumLabels; ++i) {
214 m_aLabelControls[i].SetShow(FALSE);
215 }
216
217 // Turn on just what we need for this screen.
218 SHowToPlayPageDef* pDef = &(gs_aPageDefs[m_eCurrPage]);
219
220 if (pDef->m_iTextControlIndex != eHowToPlay_TextNone) {
221 // Replace button identifiers in the text with actual button images.
222 std::wstring replacedText =
223 app.FormatHTMLString(m_iPad, app.GetString(pDef->m_iTextStringID));
224
225 // 4J-PB - replace the title with the platform specific title, and the
226 // platform name
227 replacedText = replaceAll(replacedText, L"{*PLATFORM_NAME*}",
228 app.GetString(IDS_PLATFORM_NAME));
229 replacedText = replaceAll(replacedText, L"{*BACK_BUTTON*}",
230 app.GetString(IDS_BACK_BUTTON));
231 replacedText =
232 replaceAll(replacedText, L"{*DISABLES_ACHIEVEMENTS*}",
233 app.GetString(IDS_HOST_OPTION_DISABLES_ACHIEVEMENTS));
234
235 // Set the text colour
236 std::wstring finalText(replacedText.c_str());
237 wchar_t startTags[64];
238 swprintf(startTags, 64, L"<font color=\"#%08x\">",
239 app.GetHTMLColour(eHTMLColor_White));
240 finalText = startTags + finalText;
241
242 // Set the text in the xui scene.
243 m_aTextControls[pDef->m_iTextControlIndex].SetText(finalText.c_str());
244
245 // Make it visible.
246 m_aTextControls[pDef->m_iTextControlIndex].SetShow(TRUE);
247
248 XuiElementSetUserFocus(
249 m_aTextControls[pDef->m_iTextControlIndex].m_hObj, m_iPad);
250 }
251
252 if (pDef->m_iLabelCount != 0) {
253 for (int i = pDef->m_iLabelStartIndex;
254 i < (pDef->m_iLabelStartIndex + pDef->m_iLabelCount); i++) {
255 m_aLabelControls[i].SetShow(TRUE);
256 }
257 }
258
259 if (pDef->m_iImageControlIndex != eHowToPlay_ImageNone) {
260 m_aImageControls[pDef->m_iImageControlIndex].SetShow(TRUE);
261 }
262
263 // Tool tips.
264 int iPage = (int)(m_eCurrPage);
265 if (iPage == 0) {
266 // No previous page.
267 ui.SetTooltips(iBaseSceneUser, IDS_HOW_TO_PLAY_NEXT, IDS_TOOLTIPS_BACK,
268 -1);
269 } else if ((iPage + 1) == eHowToPlay_NumPages) {
270 // No next page.
271 ui.SetTooltips(iBaseSceneUser, -1, IDS_TOOLTIPS_BACK,
272 IDS_HOW_TO_PLAY_PREV);
273 } else {
274 ui.SetTooltips(iBaseSceneUser, IDS_HOW_TO_PLAY_NEXT, IDS_TOOLTIPS_BACK,
275 IDS_HOW_TO_PLAY_PREV);
276 }
277
278 TelemetryManager->RecordMenuShown(m_iPad, eUIScene_HowToPlay,
279 (ETelemetry_HowToPlay_SubMenuId)ePage);
280}
