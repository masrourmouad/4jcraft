36void getSaveInfoReturnCallback(void* lpParam, SonyRemoteStorage::Status s,
37 int error_code) {
38 SonyRemoteStorage* pRemoteStorage = (SonyRemoteStorage*)lpParam;
39 app.DebugPrintf("remoteStorageGetInfoCallback err : 0x%08x\n", error_code);
40 if (error_code == 0) {
41 for (int i = 0; i < statParams.numFiles; i++) {
42 if (strcmp(statParams.data[i].fileName, sc_remoteSaveFilename) ==
43 0) {
44 // found the file we need in the cloud
45 pRemoteStorage->m_getInfoStatus =
46 SonyRemoteStorage::e_infoFound;
47 pRemoteStorage->m_remoteFileInfo = &statParams.data[i];
48 }
49 }
50 }
51 if (pRemoteStorage->m_getInfoStatus != SonyRemoteStorage::e_infoFound)
52 pRemoteStorage->m_getInfoStatus = SonyRemoteStorage::e_noInfoFound;
53}
